## 1. Goals

1. Student registration with a unique number `ABECFX/YY/NNNN` (YY = admission year, NNNN = student number), saved in a Google Sheet, deployed on Render.
2. Registration fields: country, state/region, programme (Forex Mentorship), full name, email, WhatsApp number, gender, passport photo, optional promo code.
3. Top-of-the-range branding and UI using the purple bull-horn logo.
4. Student status managed by the sheet editor: Uncertified (default), Certified, Graduate, Alumni Member, Mentor, Staff Member, CEO, with a status date.
5. Public "find a student" page that shows a student ID card: QR code to the live status, national flag, passport photo, graduate stamp, expiry.
6. Student portal as the home page so returning students land on their ID card.
7. Telegram quiz bot connected to the portal: student enters registration number, gets their name, takes the full quiz, sees the result with an A-F grade in the portal.
8. Editor dashboard: student statistics and quiz results.
9. Keep the free Render service from sleeping.
10. Keep this status file current with every code change.

```js
function admissionYY() {
  const y = new Intl.DateTimeFormat("en-GB", {
    timeZone: "Africa/Lagos",
    year: "numeric",
  }).format(new Date());
  return y.slice(-2);
}

function lagosTimestamp() {
  return new Intl.DateTimeFormat("en-GB", {
    timeZone: "Africa/Lagos",
    dateStyle: "medium",
    timeStyle: "medium",

}

// Serialise registrations so two people can never get the same number.
let queue = Promise.resolve();
function exclusive(task) {
  const run = queue.then(task, task);
  queue = run.catch(() => {});
  return run;
}

function clean(v, max = 120) {
  return String(v == null ? "" : v).replace(/\s+/g, " ").trim().slice(0, max);
}

// The browser shrinks the photo to 320x400 JPEG (about 20-35 KB). Accept only that kind of file.
function checkPhoto(p) {
  if (typeof p !== "string" || p.length < 1500 || p.length > 46000) return null;
  if (!/^[A-Za-z0-9+/]+={0,2}$/.test(p)) return null;
  const head = Buffer.from(p.slice(0, 8), "base64");
  if (!(head[0] === 0xff && head[1] === 0xd8 && head[2] === 0xff)) return null; // JPEG signature
  return p;
}

function validate(body) {
  const d = {
    country: clean(body.country),
    state: clean(body.state, 60),
    programme: clean(body.programme),
    fullName: clean(body.fullName),
    email: clean(body.email, 254).toLowerCase(),
    whatsapp: clean(body.whatsapp, 25),
    gender: clean(body.gender),
    promoCode: clean(body.promoCode, 30).toUpperCase(),
    photo: checkPhoto(body.photo),
  };
  const errors = {};
  if (!countries.includes(d.country)) errors.country = "Please select your country.";
  if (!errors.country) {
    const list = states[d.country];
    if (list) {
      if (!list.includes(d.state)) errors.state = "Please select your state or region.";
    } else if (d.state.length < 2) {
      errors.state = "Please enter your state, province or region.";
    }
  }
  if (!PROGRAMMES.includes(d.programme)) errors.programme = "Please select a programme.";
  if (d.fullName.length < 3 || !/\s/.test(d.fullName))
    errors.fullName = "Please enter your full name (first and last name).";
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(d.email))
    errors.email = "Please enter a valid email address.";
  const digits = d.whatsapp.replace(/[^\d]/g, "");
  if (!/^\+?[\d\s\-()]+$/.test(d.whatsapp) || digits.length < 7 || digits.length > 15)
    errors.whatsapp = "Enter a valid WhatsApp number with country code, e.g. +234 801 234 5678.";
  if (!GENDERS.includes(d.gender)) errors.gender = "Please select your gender.";
  if (d.promoCode && !/^[A-Z0-9_-]{2,30}$/.test(d.promoCode))
    errors.promoCode = "Promo code can only contain letters, numbers, - and _.";
  if (!d.photo) errors.photo = "Add a passport photograph. Choose a clear photo of your face.";
  return { d, errors };
}

// What the public ID card is allowed to show. Email and WhatsApp number are never included.
function cardFromRow(r) {
  const status = store.STATUSES.includes(r[9]) ? r[9] : "Uncertified";
  const reg = r[0] || "";
  const yy = parseInt((reg.split("/")[1] || ""), 10);
  const issuedISO = idrules.parseIssued(r);
  const expiry = idrules.expiryInfo(status, issuedISO, idrules.lagosToday());
  return {
    regNumber: reg,
    fullName: r[4] || "",
    gender: r[7] || "",
    country: r[2] || "",
    state: r[12] || "",
    countryCode: countryCodes[r[2]] || "",
    programme: r[3] || "",
    admitted: Number.isNaN(yy) ? "" : 2000 + yy,
    issued: issuedISO ? idrules.formatISO(issuedISO) : (r[1] || "").split(",")[0],
    status,
    statusDate: status === "Uncertified" ? "" : r[10] || "",
    ...expiry, // expiryKind, expires, expired, daysLeft
  };
}

function safeEqual(a, b) {
  const h = (v) => crypto.createHash("sha256").update(String(v)).digest();
  return crypto.timingSafeEqual(h(a), h(b));
}

// The quiz bot proves who it is with a shared secret in the x-quiz-key header (QUIZ_API_KEY).
function botAuth(req, res, next) {
  const key = process.env.QUIZ_API_KEY;
  if (!key) return res.status(503).json({ error: "The quiz link is not set up on the website yet." });
  if (!safeEqual(req.get("x-quiz-key") || "", key)) return res.status(401).json({ error: "Not allowed." });
  next();
}

function adminAuth(req, res, next) {
  const pw = process.env.ADMIN_PASSWORD;
  if (!pw) return res.status(503).json({ error: "The editor dashboard is not set up yet. Set ADMIN_PASSWORD on Render." });
  const given = (req.get("authorization") || "").replace(/^Bearer\s+/i, "");
  if (!safeEqual(given, pw)) return res.status(401).json({ error: "Wrong password." });
  next();
}

// Quiz bot: save a finished quiz. The grade is worked out here, not trusted from the bot.
app.post("/api/quiz/result", botAuth, async (req, res) => {
  const b = req.body || {};
  const reg = normaliseReg(b.reg);
  const score = Number(b.score);
  const total = Number(b.total);
  const tgId = clean(b.telegramId, 20);
  if (!reg || !Number.isInteger(score) || !Number.isInteger(total) || total < 1 || total > 500 || score < 0 || score > total) {
    return res.status(400).json({ error: "That result could not be read." });
  }
  if (!/^\d{1,20}$/.test(tgId)) return res.status(400).json({ error: "Missing Telegram account." });
  let breakdown = "";
  try { breakdown = JSON.stringify(b.breakdown || {}).slice(0, 3000); } catch (e) { breakdown = ""; }

  try {
    const out = await exclusive(async () => {
      const students = await store.readAll();
      const stu = students.find((r) => (r[0] || "").toUpperCase() === reg);
      if (!stu) return { status: 404, body: { error: "No student found with that registration number." } };

      const quizRows = await store.readQuiz(0);
      const mine = quizRows.filter((q) => q[1] === reg);
      // First Telegram account to take the quiz for a number keeps it, so others can't post results under it.
      const owner = mine.find((q) => q[8]);
      if (owner && owner[8] !== tgId) {
        return { status: 403, body: { error: "This registration number is already linked to another Telegram account. Contact ABECFX." } };
      }
      // Ignore an accidental double submit (same result within 2 minutes).
      const prev = mine[mine.length - 1];
      if (prev && Number(prev[3]) === score && Number(prev[4]) === total && prev[10] &&
          Date.now() - Date.parse(prev[10]) < 120000) {
```

#### `idrules.js`

```js
// ID card rules: who gets an expiry date, and the date maths behind it.
// Certified students are graduates. Only Uncertified students (not yet graduates) have IDs that are
// valid for 2 months from the issue date. Certified, Graduate, Alumni Member, Mentor, Staff Member
// and CEO have no expiry.

const MONTHS = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
const EXPIRING_STATUSES = ["Uncertified"];
const VALID_MONTHS = 2;
const ISO = /^\d{4}-\d{2}-\d{2}$/;

const pad = (n) => String(n).padStart(2, "0");

// Today's date in Lagos as YYYY-MM-DD.
function lagosToday(now = new Date()) {
  return new Intl.DateTimeFormat("en-CA", { timeZone: "Africa/Lagos" }).format(now);
}

// Issue date as YYYY-MM-DD. Uses the "ID Issue Date" column when it holds one; otherwise reads the
// "13 Oct 2026, 14:20:11" text in "Registered At", so students registered before this column existed still work.
function parseIssued(row) {
  const explicit = String(row[11] || "").trim();
  if (ISO.test(explicit)) return explicit;
  return parseRegistered(row);
}

// Date the student joined, as YYYY-MM-DD, read from "Registered At" ("13 Oct 2026, 14:20:11").
// Unlike the ID issue date, this never changes when an editor renews an ID.
function parseRegistered(row) {
  const m = /^(\d{1,2})\s+([A-Za-z]{3,})\.?\s+(\d{4})/.exec(String(row[1] || "").trim());
  if (m) {
    const mi = MONTHS.findIndex((x) => x.toLowerCase() === m[2].slice(0, 3).toLowerCase());
    if (mi >= 0) return `${m[3]}-${pad(mi + 1)}-${pad(Number(m[1]))}`;
  }
  return "";
}

// Add whole months, keeping the day of month where possible (31 Dec + 2 months = 28/29 Feb).
function addMonthsISO(iso, n) {
  const [y, m, d] = iso.split("-").map(Number);
  const t = m - 1 + n;
  const ny = y + Math.floor(t / 12);
  const nm = ((t % 12) + 12) % 12;
  const lastDay = new Date(Date.UTC(ny, nm + 1, 0)).getUTCDate();
  return `${ny}-${pad(nm + 1)}-${pad(Math.min(d, lastDay))}`;
}

function formatISO(iso) {
  const [y, m, d] = iso.split("-").map(Number);
  return `${pad(d)} ${MONTHS[m - 1]} ${y}`;
}

function daysBetween(fromIso, toIso) {
  return Math.round((Date.parse(toIso + "T00:00:00Z") - Date.parse(fromIso + "T00:00:00Z")) / 86400000);
}

// expiryKind: "none" (never expires), "date" (has an expiry date), "unknown" (should expire but no issue date).
// An ID is valid through its expiry date and counts as expired from the day after.
function expiryInfo(status, issuedISO, todayISO) {
  if (!EXPIRING_STATUSES.includes(status)) {
    return { expiryKind: "none", expires: "", expired: false, daysLeft: null };
  }
  if (!issuedISO) return { expiryKind: "unknown", expires: "", expired: false, daysLeft: null };
  const exp = addMonthsISO(issuedISO, VALID_MONTHS);
  const daysLeft = daysBetween(todayISO, exp);
  return { expiryKind: "date", expires: formatISO(exp), expired: daysLeft < 0, daysLeft };
}

module.exports = { lagosToday, parseIssued, parseRegistered, addMonthsISO, formatISO, daysBetween, expiryInfo, EXPIRING_STATUSES };
```

```js
// A-F grading scale for quiz results. Change the numbers here and the whole site follows.
// Scale used (common Nigerian grading): A 70-100, B 60-69, C 50-59, D 45-49, E 40-44, F 0-39.
const SCALE = [
  { grade: "A", min: 70, label: "Excellent" },
  { grade: "B", min: 60, label: "Very good" },

  { grade: "D", min: 45, label: "Fair" },
  { grade: "E", min: 40, label: "Pass" },
  { grade: "F", min: 0, label: "Fail" },
];

// pct is the whole-number percentage that is shown to the student (0-100).
function gradeFor(pct) {
  const p = Number.isFinite(pct) ? pct : 0;
  return SCALE.find((s) => p >= s.min) || SCALE[SCALE.length - 1];
}

function percent(score, total) {
  return total > 0 ? Math.round((score / total) * 100) : 0;
}
```

```js
// Statistics for the editor dashboard. Pure functions: rows in, numbers out.
// Student row columns: 0 reg, 1 registered at, 2 country, 3 programme, 4 name, 7 gender, 9 status, 12 state.
// Quiz row columns: 0 submitted at, 1 reg, 2 name, 3 score, 4 total, 5 percent, 6 grade, 7 attempt.
const idrules = require("./idrules");
const grading = require("./grading");

const GRADUATE_STATUSES = ["Certified", "Graduate", "Alumni Member"];
const TEAM_STATUSES = ["Mentor", "Staff Member", "CEO"];
const STATUS_ORDER = ["Uncertified", "Certified", "Graduate", "Alumni Member", "Mentor", "Staff Member", "CEO"];

function tally(values, blank = "Not recorded") {
  const m = new Map();
  for (const v of values) {
    const k = (v || "").trim() || blank;
    m.set(k, (m.get(k) || 0) + 1);
  }
  return [...m.entries()]
    .map(([label, count]) => ({ label, count }))
    .sort((a, b) => b.count - a.count || a.label.localeCompare(b.label));
}

function ymOf(iso) {
  return iso.slice(0, 7);
}

function nextYm(ym) {
  let [y, m] = ym.split("-").map(Number);
  m += 1;
  if (m > 12) { m = 1; y += 1; }
  return `${y}-${String(m).padStart(2, "0")}`;
}

function statusOf(r) {
  return STATUS_ORDER.includes(r[9]) ? r[9] : "Uncertified";
}

function buildStats(rows, quizRows, todayISO) {
  const total = rows.length;
  const statusCounts = Object.fromEntries(STATUS_ORDER.map((s) => [s, 0]));
  for (const r of rows) statusCounts[statusOf(r)] += 1;
  const sum = (list) => list.reduce((n, s) => n + statusCounts[s], 0);

  // ---- joining dates ----
  const joined = rows.map((r) => idrules.parseRegistered(r)).filter(Boolean);
  const undated = total - joined.length;
  const perMonth = new Map();
  for (const iso of joined) perMonth.set(ymOf(iso), (perMonth.get(ymOf(iso)) || 0) + 1);

  const months = [];
  if (perMonth.size) {
    const sorted = [...perMonth.keys()].sort();
    const last = [sorted[sorted.length - 1], ymOf(todayISO)].sort().pop();
    let running = 0;
    let prevNew = null;
    for (let ym = sorted[0]; ym <= last; ym = nextYm(ym)) {
      const n = perMonth.get(ym) || 0;
      running += n;
      months.push({
        ym,
        new: n,
        total: running,
        growthPct: prevNew === null || prevNew === 0 ? null : Math.round(((n - prevNew) / prevNew) * 100),
      });
      prevNew = n;
    }
  }
  const monthsByYear = {};
  for (const m of months) {
    const [y, mo] = m.ym.split("-");
    (monthsByYear[y] = monthsByYear[y] || Array(12).fill(0))[Number(mo) - 1] = m.new;
  }

  // year joined: the registration number's year, which is the admission year
  const byYear = tally(
    rows.map((r) => {
      const yy = parseInt(String(r[0] || "").split("/")[1], 10);
      return Number.isNaN(yy) ? "" : String(2000 + yy);
    })
  ).sort((a, b) => a.label.localeCompare(b.label));

  // ---- quiz ----
  const latest = new Map(); // reg -> most recent attempt
  const best = new Map();
  for (const q of quizRows) {
    const reg = q[1];
    const pct = Number(q[5]) || 0;
    latest.set(reg, q);
    if (!best.has(reg) || pct > (Number(best.get(reg)[5]) || 0)) best.set(reg, q);
  }
  const latestList = [...latest.values()];
  const gradeDist = grading.SCALE.map((s) => ({
    grade: s.grade,
    label: s.label,
    count: latestList.filter((q) => q[6] === s.grade).length,
  }));
  const avgPct = latestList.length
    ? Math.round(latestList.reduce((n, q) => n + (Number(q[5]) || 0), 0) / latestList.length)
    : null;

  return {
    total,
    gender: tally(rows.map((r) => r[7])),
    country: tally(rows.map((r) => r[2])),
    state: tally(rows.map((r) => (r[12] ? `${r[12]}${r[2] ? ", " + r[2] : ""}` : ""))),
    programme: tally(rows.map((r) => r[3])),
    status: STATUS_ORDER.map((s) => ({ label: s, count: statusCounts[s] })),
    graduates: sum(GRADUATE_STATUSES), // Certified + Graduate + Alumni Member
    uncertified: statusCounts.Uncertified,
    team: sum(TEAM_STATUSES), // Mentor + Staff Member + CEO
    byYear,
    months,
    monthsByYear,
    undated,
    quiz: {
      attempts: quizRows.length,
      students: latest.size,
      notTaken: Math.max(0, total - latest.size),
      avgPct,
      gradeDist,
    },
  };
}

module.exports = { buildStats, GRADUATE_STATUSES, TEAM_STATUSES };
```

```js
// Keeps a Render free instance from going idle: while the app is running it requests its own public
// /healthz address every 10 minutes. Render counts that as traffic, so the 15-minute idle timer resets.
// It cannot wake a sleeping service, so pair it with an outside pinger (see README).
// Render sets RENDER_EXTERNAL_URL automatically. Set KEEP_AWAKE=off to turn this off.
function startKeepAlive({ intervalMs = 10 * 60 * 1000, url = process.env.RENDER_EXTERNAL_URL, log = console } = {}) {
  if (process.env.KEEP_AWAKE === "off" || !url) return null;
  const target = url.replace(/\/+$/, "") + "/healthz";
  const ping = async () => {
    try {
      const res = await fetch(target, { signal: AbortSignal.timeout(15000) });
      if (!res.ok) log.error("Keep-alive ping got status", res.status);
    } catch (e) {
      log.error("Keep-alive ping failed:", e.message);
    }
  };
  const timer = setInterval(ping, intervalMs);
  timer.unref();
  return { timer, ping, target };
}

module.exports = { startKeepAlive };
```

```js
  function openViaReader(file) {
    return new Promise(function (resolve, reject) {
      var fr = new FileReader();
      fr.onerror = function () { reject(new Error("read")); };
      fr.onload = function () {
        var img = new Image();
        img.onload = function () { resolve({ src: img, w: img.naturalWidth, h: img.naturalHeight }); };
        img.onerror = function () { reject(new Error("decode")); };
```

```js
/*
 * Minimal QR Code generator (byte mode, error correction level M, versions 1-10).
 * Written for the ABECFX ID card so the site needs no outside QR service.
 * Exposes: QR.encode(text) -> { size, modules[y][x] }  and  QR.toSvg(text, { border })
 */
(function () {
  "use strict";

  var EC_PER_BLOCK = [0, 10, 16, 26, 18, 24, 16, 18, 22, 22, 26]; // level M
  var NUM_BLOCKS = [0, 1, 1, 1, 2, 2, 4, 4, 4, 5, 5];             // level M
  var MAX_VERSION = 10;

  function rawModules(ver) {
    var r = (16 * ver + 128) * ver + 64;
    if (ver >= 2) {
      var n = Math.floor(ver / 7) + 2;
      r -= (25 * n - 10) * n - 55;
      if (ver >= 7) r -= 36;
    }
    return r;
  }
  function dataCodewords(ver) {
    return Math.floor(rawModules(ver) / 8) - EC_PER_BLOCK[ver] * NUM_BLOCKS[ver];
  }

  // ---- Reed-Solomon over GF(256), polynomial 0x11D ----
  function gfMul(x, y) {
    var z = 0;
    for (var i = 7; i >= 0; i--) {
      z = (z << 1) ^ ((z >>> 7) * 0x11d);
      z ^= ((y >>> i) & 1) * x;
    }
    return z;
  }
  function rsDivisor(degree) {
    var result = [], i, j;
    for (i = 0; i < degree - 1; i++) result.push(0);
    result.push(1);
    var root = 1;
    for (i = 0; i < degree; i++) {
      for (j = 0; j < result.length; j++) {
        result[j] = gfMul(result[j], root);
        if (j + 1 < result.length) result[j] ^= result[j + 1];
      }
      root = gfMul(root, 2);
    }
    return result;
  }
  function rsRemainder(data, divisor) {
    var result = divisor.map(function () { return 0; });
    data.forEach(function (b) {
      var factor = b ^ result.shift();
      result.push(0);
      divisor.forEach(function (coef, i) { result[i] ^= gfMul(coef, factor); });
    });
    return result;
  }

  function addEccAndInterleave(data, ver) {
    var numBlocks = NUM_BLOCKS[ver];
    var eccLen = EC_PER_BLOCK[ver];
    var raw = Math.floor(rawModules(ver) / 8);
    var numShort = numBlocks - (raw % numBlocks);
    var shortLen = Math.floor(raw / numBlocks);
    var blocks = [];
    var div = rsDivisor(eccLen);
    for (var i = 0, k = 0; i < numBlocks; i++) {
      var dat = data.slice(k, k + shortLen - eccLen + (i < numShort ? 0 : 1));
      k += dat.length;
      var ecc = rsRemainder(dat, div);
      if (i < numShort) dat.push(0);
      blocks.push(dat.concat(ecc));
    }
    var result = [];
    for (var c = 0; c < blocks[0].length; c++) {
      blocks.forEach(function (block, j) {
        if (c !== shortLen - eccLen || j >= numShort) result.push(block[c]);
      });
    }
    return result;
  }

  function utf8Bytes(str) {
    var s = unescape(encodeURIComponent(str)), out = [];
    for (var i = 0; i < s.length; i++) out.push(s.charCodeAt(i));
    return out;
  }
  function getBit(x, i) { return ((x >>> i) & 1) !== 0; }

  function encode(text) {
    var bytes = utf8Bytes(text);
    var ver, capBits;
    for (ver = 1; ver <= MAX_VERSION; ver++) {
      capBits = dataCodewords(ver) * 8;
      var ccBits = ver < 10 ? 8 : 16;
      if (4 + ccBits + bytes.length * 8 <= capBits) break;
    }
    if (ver > MAX_VERSION) throw new Error("QR text too long");

    var bits = [];
    function push(val, len) { for (var i = len - 1; i >= 0; i--) bits.push((val >>> i) & 1); }
    push(4, 4);
    push(bytes.length, ver < 10 ? 8 : 16);
    bytes.forEach(function (b) { push(b, 8); });
    push(0, Math.min(4, capBits - bits.length));
    push(0, (8 - (bits.length % 8)) % 8);
    for (var pad = 0xec; bits.length < capBits; pad ^= 0xec ^ 0x11) push(pad, 8);

    var data = [];
    for (var i = 0; i < bits.length; i += 8) {
      var v = 0;
      for (var j = 0; j < 8; j++) v = (v << 1) | bits[i + j];
      data.push(v);
    }
    var all = addEccAndInterleave(data, ver);

    var size = ver * 4 + 17;
    var modules = [], isFn = [];
    for (var y = 0; y < size; y++) {
      modules.push(new Array(size).fill(false));
      isFn.push(new Array(size).fill(false));
    }
    function setFn(x, y, dark) { modules[y][x] = dark; isFn[y][x] = true; }

    function drawFinder(cx, cy) {
      for (var dy = -4; dy <= 4; dy++) {
        for (var dx = -4; dx <= 4; dx++) {
          var dist = Math.max(Math.abs(dx), Math.abs(dy));
          var xx = cx + dx, yy = cy + dy;
          if (xx >= 0 && xx < size && yy >= 0 && yy < size) setFn(xx, yy, dist !== 2 && dist !== 4);
        }
      }
    }
    function drawAlign(cx, cy) {
      for (var dy = -2; dy <= 2; dy++)
        for (var dx = -2; dx <= 2; dx++)
          setFn(cx + dx, cy + dy, Math.max(Math.abs(dx), Math.abs(dy)) !== 1);
    }
    function alignPositions() {
      if (ver === 1) return [];
      var n = Math.floor(ver / 7) + 2;
      var step = Math.ceil((ver * 4 + 4) / (n * 2 - 2)) * 2;
      var res = [6];
      for (var pos = size - 7; res.length < n; pos -= step) res.splice(1, 0, pos);
      return res;
    }
    function drawFormat(mask) {
      var d = (0 << 3) | mask; // level M = 00
      var rem = d, i;
      for (i = 0; i < 10; i++) rem = (rem << 1) ^ ((rem >>> 9) * 0x537);
      var b = ((d << 10) | rem) ^ 0x5412;
      for (i = 0; i <= 5; i++) setFn(8, i, getBit(b, i));
      setFn(8, 7, getBit(b, 6));
      setFn(8, 8, getBit(b, 7));
      setFn(7, 8, getBit(b, 8));
      for (i = 9; i < 15; i++) setFn(14 - i, 8, getBit(b, i));
      for (i = 0; i < 8; i++) setFn(size - 1 - i, 8, getBit(b, i));
      for (i = 8; i < 15; i++) setFn(8, size - 15 + i, getBit(b, i));
      setFn(8, size - 8, true);
    }
    function drawVersion() {
      if (ver < 7) return;
      var rem = ver, i;
      for (i = 0; i < 12; i++) rem = (rem << 1) ^ ((rem >>> 11) * 0x1f25);
      var b = (ver << 12) | rem;
      for (i = 0; i < 18; i++) {
        var bit = getBit(b, i), a = size - 11 + (i % 3), c = Math.floor(i / 3);
        setFn(a, c, bit);
        setFn(c, a, bit);
      }
    }

    for (i = 0; i < size; i++) { setFn(6, i, i % 2 === 0); setFn(i, 6, i % 2 === 0); }
    drawFinder(3, 3); drawFinder(size - 4, 3); drawFinder(3, size - 4);
    var ap = alignPositions(), last = ap.length - 1;
    for (i = 0; i < ap.length; i++) {
      for (j = 0; j < ap.length; j++) {
        if ((i === 0 && j === 0) || (i === 0 && j === last) || (i === last && j === 0)) continue;
        drawAlign(ap[i], ap[j]);
      }
    }
    var j;
    drawFormat(0);
    drawVersion();

    // place codewords in the zig-zag order
    var n = 0;
    for (var right = size - 1; right >= 1; right -= 2) {
      if (right === 6) right = 5;
      for (var vert = 0; vert < size; vert++) {
        for (j = 0; j < 2; j++) {
          var x = right - j;
          var upward = ((right + 1) & 2) === 0;
          var yy = upward ? size - 1 - vert : vert;
          if (!isFn[yy][x] && n < all.length * 8) {
            modules[yy][x] = getBit(all[n >>> 3], 7 - (n & 7));
            n++;
          }
        }
      }
    }

    function applyMask(mask) {
      for (var y = 0; y < size; y++) {
        for (var x = 0; x < size; x++) {
          var inv;
          switch (mask) {
            case 0: inv = (x + y) % 2 === 0; break;
            case 1: inv = y % 2 === 0; break;
            case 2: inv = x % 3 === 0; break;
            case 3: inv = (x + y) % 3 === 0; break;
            case 4: inv = (Math.floor(x / 3) + Math.floor(y / 2)) % 2 === 0; break;
            case 5: inv = ((x * y) % 2) + ((x * y) % 3) === 0; break;
            case 6: inv = (((x * y) % 2) + ((x * y) % 3)) % 2 === 0; break;
            default: inv = (((x + y) % 2) + ((x * y) % 3)) % 2 === 0;
          }
          if (!isFn[y][x] && inv) modules[y][x] = !modules[y][x];
        }
      }
    }

    function penalty() {
      var score = 0, x, y, run, k, line;
      var dark = 0;
      for (y = 0; y < size; y++) for (x = 0; x < size; x++) if (modules[y][x]) dark++;
      function runs(get) {
        for (var a = 0; a < size; a++) {
          run = 1;
          for (var b = 1; b <= size; b++) {
            if (b < size && get(a, b) === get(a, b - 1)) run++;
            else { if (run >= 5) score += 3 + (run - 5); run = 1; }
          }
        }
      }
      runs(function (a, b) { return modules[a][b]; });
      runs(function (a, b) { return modules[b][a]; });
      for (y = 0; y < size - 1; y++)
        for (x = 0; x < size - 1; x++) {
          var c = modules[y][x];
          if (c === modules[y][x + 1] && c === modules[y + 1][x] && c === modules[y + 1][x + 1]) score += 3;
        }
      function finderLike(s) {
        var count = 0, idx = -1, pats = ["00001011101", "10111010000"];
        pats.forEach(function (p) {
          idx = s.indexOf(p);
          while (idx !== -1) { count++; idx = s.indexOf(p, idx + 1); }
        });
        return count;
      }
      for (y = 0; y < size; y++) {
        line = ""; for (x = 0; x < size; x++) line += modules[y][x] ? "1" : "0";
        score += 40 * finderLike(line);
      }
      for (x = 0; x < size; x++) {
        line = ""; for (y = 0; y < size; y++) line += modules[y][x] ? "1" : "0";
        score += 40 * finderLike(line);
      }
      k = Math.ceil(Math.abs(dark * 20 - size * size * 10) / (size * size)) - 1;
      score += Math.max(0, k) * 10;
      return score;
    }

    var best = 0, bestScore = Infinity, m;
    for (m = 0; m < 8; m++) {
      applyMask(m);
      drawFormat(m);
      var p = penalty();
      if (p < bestScore) { bestScore = p; best = m; }
      applyMask(m); // undo
    }
    applyMask(best);
    drawFormat(best);
    return { size: size, version: ver, mask: best, modules: modules };
  }

  function toSvg(text, opts) {
    var border = opts && opts.border != null ? opts.border : 2;
    var qr = encode(text);
    var ns = "http://www.w3.org/2000/svg";
    var d = "";
    for (var y = 0; y < qr.size; y++)
      for (var x = 0; x < qr.size; x++)
        if (qr.modules[y][x]) d += "M" + (x + border) + "," + (y + border) + "h1v1h-1z";
    var total = qr.size + border * 2;
    var svg = document.createElementNS(ns, "svg");
    svg.setAttribute("viewBox", "0 0 " + total + " " + total);
    svg.setAttribute("shape-rendering", "crispEdges");
    svg.setAttribute("role", "img");
    svg.setAttribute("aria-label", "QR code to check this student's status");
    var bg = document.createElementNS(ns, "rect");
    bg.setAttribute("width", total); bg.setAttribute("height", total); bg.setAttribute("fill", "#fff");
    var path = document.createElementNS(ns, "path");
    path.setAttribute("d", d); path.setAttribute("fill", "#14081a");
    svg.appendChild(bg); svg.appendChild(path);
    return svg;
  }

  var api = { encode: encode, toSvg: toSvg };
  if (typeof module !== "undefined" && module.exports) module.exports = api;
  else window.QR = api;
})();
```

```js
    var data = (stats.monthsByYear[y] || Array(12).fill(0));
    if (!y) { $("c-monthyear").replaceChildren(el("p", "empty", "No joining dates yet.")); return; }
    columns($("c-monthyear"), MONTH.map(function (m, i) { return { label: m, value: data[i] }; }));
  }

  // ---------- Quiz ----------
  function renderQuiz() {
    var q = stats.quiz;
    $("quizKpis").replaceChildren(
      kpi("Students who took the quiz", q.students),
      kpi("Not yet taken", q.notTaken),
      kpi("Total attempts", q.attempts),
      kpi("Average score", q.avgPct == null ? "-" : q.avgPct + "%")
    );
    bars($("c-grades"), q.gradeDist.map(function (g) { return { label: g.grade + " - " + g.label, count: g.count }; }), { total: q.students });
    table($("c-scale"), [{ t: "Grade" }, { t: "Score" }, { t: "Meaning" }],
      scale.map(function (s, i) {
        var hi = i === 0 ? 100 : scale[i - 1].min - 1;
        return [gradeBadge(s.grade), s.min + " - " + hi + "%", s.label];
      }));
    var gf = $("gradeFilter");
    if (gf.options.length <= 1) scale.forEach(function (s) { var o = el("option", null, "Grade " + s.grade); o.value = s.grade; gf.appendChild(o); });
    renderResults();
  }

  function filtered() {
    var term = $("q").value.trim().toLowerCase();
    var grade = $("gradeFilter").value;
    var mode = $("viewMode").value;
    var list = results;
    if (mode !== "all") {
      var map = new Map();
      // results is newest first
      list.forEach(function (r) {
        var cur = map.get(r.regNumber);
        if (mode === "latest") { if (!cur) map.set(r.regNumber, r); }
        else if (!cur || r.pct > cur.pct) map.set(r.regNumber, r);
      });
      list = Array.from(map.values());
    }
    return list.filter(function (r) {
      if (grade && r.grade !== grade) return false;
      if (term && (r.fullName + " " + r.regNumber).toLowerCase().indexOf(term) < 0) return false;
      return true;
    });
  }

  function renderResults() {
    var list = filtered();
    table($("resultsTable"),
      [{ t: "Date" }, { t: "Reg number" }, { t: "Name" }, { t: "Score", n: 1 }, { t: "Percent", n: 1 }, { t: "Grade" }, { t: "Attempt", n: 1 }],
      list.slice(0, 500).map(function (r) {
        return [r.date, r.regNumber, r.fullName, r.score + " / " + r.total, r.pct + "%", gradeBadge(r.grade), r.attempt];
      }));
    $("resultsNote").textContent = list.length
      ? "Showing " + Math.min(list.length, 500) + " of " + list.length + " result(s). Download the CSV for the full list."
      : "No quiz results match.";
  }

  function csvCell(v) {
    var s = String(v == null ? "" : v);
    if (/^[=+\-@\t\r]/.test(s)) s = "'" + s; // stop spreadsheets running names as formulas
    return /[",\n]/.test(s) ? '"' + s.replace(/"/g, '""') + '"' : s;
  }

  function downloadCsv() {
    var head = ["Date", "Reg Number", "Name", "Score", "Total", "Percent", "Grade", "Attempt"];
    var lines = [head.join(",")].concat(filtered().map(function (r) {
      return [r.date, r.regNumber, r.fullName, r.score, r.total, r.pct, r.grade, r.attempt].map(csvCell).join(",");
    }));
    var blob = new Blob(["﻿" + lines.join("\r\n")], { type: "text/csv;charset=utf-8" });
    var a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = "abecfx-quiz-results.csv";
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(function () { URL.revokeObjectURL(a.href); }, 2000);
  }

  // ---------- Events ----------
  function setTab(name) {
    ["overview", "quiz"].forEach(function (n) {
      $("view-" + n).hidden = n !== name;
      $("tab-" + n).setAttribute("aria-selected", n === name ? "true" : "false");
    });
  }
  $("tab-overview").addEventListener("click", function () { setTab("overview"); });
  $("tab-quiz").addEventListener("click", function () { setTab("quiz"); });
  $("yearSel").addEventListener("change", drawMonthYear);
  ["q", "gradeFilter", "viewMode"].forEach(function (id) {
    $(id).addEventListener(id === "q" ? "input" : "change", renderResults);
  });
  $("csvBtn").addEventListener("click", downloadCsv);

  $("loginForm").addEventListener("submit", function (e) {
    e.preventDefault();
    var pw = $("pw").value;
    if (!pw) { $("loginErr").textContent = "Enter the password."; return; }
    password = pw;
    var b = $("loginBtn");
    b.disabled = true;
    b.textContent = "Checking...";
    $("loginErr").textContent = "";
    load()
      .catch(function (err) { password = ""; $("loginErr").textContent = err.message; })
      .then(function () { b.disabled = false; b.textContent = "Sign in"; $("pw").value = ""; });
  });

  $("refreshBtn").addEventListener("click", function () {
    var b = this;
    b.disabled = true;
    load()
      .catch(function (err) { if (err.auth) showLogin(err.message); else $("stamp").textContent = err.message; })
      .then(function () { b.disabled = false; });
  });

  $("logoutBtn").addEventListener("click", function () {
    password = "";
    save("");
    stats = null;
    showLogin("");
  });

  // Come back without retyping during the same browser session.
  var remembered = getSaved();
  if (remembered) {
    password = remembered;
    load().catch(function () { password = ""; save(""); showLogin(""); });
  }
})();
```

```js
function fence(text) {
  const runs = text.match(/`+/g) || [];
  const n = Math.max(2, ...runs.map((r) => r.length)) + 1;
  return "`".repeat(n);
}
function block(title, text, lang) {
  const f = fence(text);
  return `#### \`${title}\`\n\n${f}${lang}\n${text.replace(/\s+$/, "")}\n${f}\n`;
}
function langOf(file) {
  return LANG[file.split(".").pop()] || "";
}
function readIf(p) {
  try { return fs.readFileSync(p, "utf8"); } catch (e) { return null; }
}

let out = "### Registration website\n\n";
const present = SITE_DATA.filter((f) => fs.existsSync(path.join(root, f)))
  .map((f) => `\`${f}\` (${(fs.statSync(path.join(root, f)).size / 1024).toFixed(0)} KB)`);
out += `Not embedded (data): ${present.join(", ")}.\n\n`;
for (const f of SITE_FILES) {
  const t = readIf(path.join(root, f));
  if (t == null) { out += `#### \`${f}\`\n\n_missing_\n\n`; continue; }
  out += block(f, t, langOf(f)) + "\n";
}

out += "### Telegram quiz bot (separate repository)\n\n";
if (fs.existsSync(quizDir)) {

    const t = readIf(path.join(quizDir, f));
```

```js
// ABEC-FX — Edge Handbook Quiz · Telegram bot
//
// Runs the Telegram bot AND serves the quiz's HTML from the same process,
// so this can deploy as a single Render Web Service (or any other single
// host) — no separate static site needed. Serves the quiz as a Telegram
// Mini App (Web App) and reports the score back into the chat when the
// user finishes a quiz and taps "Send score to chat & close".
//
// Setup:
//   1. Create a bot with @BotFather on Telegram, get its token.
//   2. Copy .env.example to .env and fill in BOT_TOKEN.
//      WEBAPP_URL is usually NOT needed — see "Finding the Mini App URL"
//      below. Only set it if you deployed the webapp separately instead.
//   3. npm install
//   4. npm start
//
// The bot uses long polling for Telegram updates (no inbound webhook
// needed for that), and a tiny built-in HTTP server for everything else:
// serving the quiz page, and answering Render's health check.

require('dotenv').config();
const fs = require('fs');
const path = require('path');
const http = require('http');
const TelegramBot = require('node-telegram-bot-api');

const TOKEN = process.env.BOT_TOKEN;
const PORT = process.env.PORT;

// ---------------------------------------------------------------------
// Finding the Mini App URL
// ---------------------------------------------------------------------
// Render automatically sets RENDER_EXTERNAL_URL to this exact service's
// own public URL. Since this same process serves webapp/index.html below
// (whenever PORT is set), that IS the Mini App URL — nothing to copy
// between two services. WEBAPP_URL only needs setting by hand if you
// deployed the quiz page somewhere else instead (GitHub Pages, a
// separate static site, etc.) — in that case it overrides the guess.
const WEBAPP_URL = process.env.WEBAPP_URL || process.env.RENDER_EXTERNAL_URL;

// ---------------------------------------------------------------------
// Link to the ABECFX registration website (where students and results live)
// ---------------------------------------------------------------------
// REG_SITE_URL: the registration site, e.g. https://abecfx.onrender.com
// QUIZ_API_KEY: a long random secret. Set the SAME value on the registration site.
const crypto = require('crypto');
const REG_SITE_URL = (process.env.REG_SITE_URL || '').replace(/\/+$/, '');
const QUIZ_API_KEY = process.env.QUIZ_API_KEY || '';
if (!REG_SITE_URL || !QUIZ_API_KEY) {
  console.warn('REG_SITE_URL / QUIZ_API_KEY not set: students cannot sign in to the quiz or save results until they are.');
}

if (!TOKEN) {
  console.error('Missing BOT_TOKEN. Copy .env.example to .env and fill it in.');
  process.exit(1);
}
if (!WEBAPP_URL || !/^https:\/\//.test(WEBAPP_URL)) {
  console.error(
    'Missing or invalid WEBAPP_URL. On Render this is normally automatic ' +
    '(RENDER_EXTERNAL_URL) — if you\'re running locally or elsewhere, set ' +
    'WEBAPP_URL in .env to a public https:// URL serving webapp/index.html.'
  );
  process.exit(1);
}

const bot = new TelegramBot(TOKEN, { polling: true });

console.log('ABEC-FX quiz bot starting...');
console.log('Mini App URL:', WEBAPP_URL);

// ---------------------------------------------------------------------
// Commands
// ---------------------------------------------------------------------
bot.setMyCommands([
  { command: 'start', description: 'Open the ABEC-FX Edge Handbook quiz' },
  { command: 'quiz', description: 'Open the quiz' },
  { command: 'help', description: 'What this bot does' },
]).catch((e) => console.error('setMyCommands failed:', e.message));

// Persistent "menu button" next to the message box (tap it to open the quiz
// any time, without needing /start). This sets the DEFAULT button for all
// private chats with the bot.
bot.setChatMenuButton({
  menu_button: JSON.stringify({
    type: 'web_app',
    text: 'Open Quiz',
    web_app: { url: WEBAPP_URL },
  }),
}).catch((e) => console.error('setChatMenuButton failed:', e.message));

function sendQuizButton(chatId) {
  return bot.sendMessage(
    chatId,
    '📈 *ABEC-FX — Edge Handbook Quiz*\n\nTest what you know: indicators, trend reading, entries, stop loss, retest zones and risk discipline.',
    {
      parse_mode: 'Markdown',
      reply_markup: {
        inline_keyboard: [[
          { text: '🧠 Open Quiz', web_app: { url: WEBAPP_URL } },
        ]],
      },
    }
  );

bot.onText(/^\/start/, (msg) => sendQuizButton(msg.chat.id));
bot.onText(/^\/quiz/, (msg) => sendQuizButton(msg.chat.id));
bot.onText(/^\/help/, (msg) => {
  bot.sendMessage(
    msg.chat.id,
    'This bot opens the ABEC-FX Edge Handbook quiz as an in-app Mini App.\n\n' +
    '• /quiz — open the quiz\n' +
    '• Tap "Open Quiz" in the menu button next to the message box, any time\n' +
    '• When you finish a quiz, tap "Send score to chat & close" to post your result here'

// Receiving the score back from the Mini App
// ---------------------------------------------------------------------
// The webapp calls Telegram.WebApp.sendData(JSON.stringify(payload)) when
// the user taps "Send score to chat & close". Telegram delivers that as a
// normal message with a `web_app_data` field, then closes the Mini App.
bot.on('message', (msg) => {
  if (!msg.web_app_data) return;

  let payload;
  try {
    payload = JSON.parse(msg.web_app_data.data);
  } catch (e) {
    bot.sendMessage(msg.chat.id, '🎯 Quiz finished! (Could not parse the result details.)');
    return;
  }

  if (payload.type !== 'abecfx_quiz_result') return;

  const { name, score, total, pct, verdict, missedCount, grade } = payload;
  const emoji = pct >= 80 ? '🟢' : pct >= 60 ? '🟡' : '🔴';
  const who = name ? `*${name}*` : 'You';

  let text = `${emoji} *Quiz complete*\n\n` +
    `${who} scored *${score} / ${total}* (${pct}%)${grade ? ` — Grade *${grade}*` : ''}\n` +
    `${verdict || ''}\n`;

  if (missedCount > 0) {
    text += `\nMissed: ${missedCount} question${missedCount === 1 ? '' : 's'} — open the quiz again and use "Retry missed only" to drill those.`;
  }

  bot.sendMessage(msg.chat.id, text, { parse_mode: 'Markdown' });

  }

  // ---- Telegram sign-in check -------------------------------------------------
  // The Mini App sends Telegram's signed "initData". Verifying it proves the request really comes
  // from a Telegram user, and tells us their Telegram id.
  function verifyInitData(initData) {
    if (!initData || initData.length > 4096) return null;
    const params = new URLSearchParams(initData);
    const hash = params.get('hash');
    if (!hash) return null;
    params.delete('hash');
    const check = [...params.entries()].map(([k, v]) => `${k}=${v}`).sort().join('\n');
    const secret = crypto.createHmac('sha256', 'WebAppData').update(TOKEN).digest();
    const calc = crypto.createHmac('sha256', secret).update(check).digest('hex');

    const age = Date.now() / 1000 - Number(params.get('auth_date') || 0);
    if (!(age >= 0 && age < 24 * 3600)) return null;
    try {
      const user = JSON.parse(params.get('user') || '{}');
      return user && user.id ? { id: String(user.id) } : null;
    } catch (e) { return null; }
  }

  // Small per-Telegram-account limit so nobody can hammer the registration site through the bot.
  const hits = new Map();
  function limited(id) {
    const now = Date.now();
    const arr = (hits.get(id) || []).filter((t) => now - t < 10 * 60 * 1000);
    arr.push(now);
    hits.set(id, arr);
    if (hits.size > 5000) hits.clear();
    return arr.length > 40;
  }

  function sendJson(res, status, body) {
    res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', 'Cache-Control': 'no-store' });
    res.end(JSON.stringify(body));
  }

  function readJson(req) {
    return new Promise((resolve, reject) => {
      let size = 0;
      let tooBig = false;
      const chunks = [];
      req.on('data', (c) => {
        size += c.length;
        if (tooBig) return;
        if (size > 20000) { tooBig = true; reject(new Error('too large')); return; }
        chunks.push(c);
      });
      req.on('end', () => {
        if (tooBig) return;
        try { resolve(JSON.parse(Buffer.concat(chunks).toString('utf8') || '{}')); } catch (e) { reject(e); }
      });
      req.on('error', reject);
    });
  }

  // Forward to the registration site with the shared secret.
  async function callSite(pathAndQuery, options = {}) {
    const r = await fetch(REG_SITE_URL + pathAndQuery, {
      ...options,
      headers: { 'x-quiz-key': QUIZ_API_KEY, 'Content-Type': 'application/json' },
      signal: AbortSignal.timeout(20000),
    });
    let body = {};
    try { body = await r.json(); } catch (e) { /* non-JSON error page */ }
    return { status: r.status, body };
  }

  async function handleApi(req, res, url) {
    if (!REG_SITE_URL || !QUIZ_API_KEY) {
      return sendJson(res, 503, { error: 'The quiz is not connected to the student records yet.' });
    }
    const user = verifyInitData(req.headers['x-telegram-init-data']);
    if (!user) {
      return sendJson(res, 401, { error: 'Open this quiz from the Telegram bot so we can sign you in.' });
    }
    if (limited(user.id)) return sendJson(res, 429, { error: 'Too many requests. Please wait a few minutes.' });

    try {
      if (req.method === 'GET' && url.pathname === '/api/student') {
        const out = await callSite('/api/quiz/student?reg=' + encodeURIComponent(url.searchParams.get('reg') || ''));
        return sendJson(res, out.status, out.body);
      }
      if (req.method === 'POST' && url.pathname === '/api/submit') {
        const b = await readJson(req);
        const out = await callSite('/api/quiz/result', {
          method: 'POST',
          body: JSON.stringify({
            reg: b.reg, score: b.score, total: b.total, breakdown: b.breakdown, telegramId: user.id,
          }),
        });
        return sendJson(res, out.status, out.body);
      }
      return sendJson(res, 404, { error: 'Not found.' });
    } catch (e) {
      if (e.message === 'too large') return sendJson(res, 413, { error: 'That request is too large.' });
      console.error('API error:', e.message);
      return sendJson(res, 502, { error: 'Could not reach the student records. Please try again shortly.' });
    }
  }

  http
    .createServer((req, res) => {
      const url = new URL(req.url, 'http://localhost');
      if (url.pathname.startsWith('/api/')) {
        handleApi(req, res, url);
        return;
      }
      if (url.pathname === '/healthz') {
        res.writeHead(200, { 'Content-Type': 'text/plain' });
        res.end('ok');
        return;
      }
      if (html) {
        res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
        res.end(html);
      } else {
        res.writeHead(200, { 'Content-Type': 'text/plain' });
        res.end('ABEC-FX quiz bot is running, but webapp/index.html was not found.\n');
      }
    })
    .listen(PORT, () => {
      console.log(`Serving the quiz page on port ${PORT} — this is also the Mini App URL.`);
    });
}
```
